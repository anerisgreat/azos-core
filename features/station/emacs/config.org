** Mail

*** General instructions

Prerequisites: =pass= entries set up, mbsync run at least once. Gmail accounts
require an App Password (myaccount.google.com/apppasswords) — not your regular password.

=mu= is auto-initialized by home-manager activation on first deploy. To re-initialize manually:

#+begin_src bash :results none :exports code
mu init --maildir=~/Mail \
  --my-address=aner@zakobar.com \
  --my-address=anerisgreat@gmail.com \
  --my-address=anerz@post.bgu.ac.il
mu index
#+end_src

#+begin_src emacs-lisp
(defvar azos/mail/blacklist-addresses nil
  "List of email addresses (as strings) to blacklist for
  `mml-secure-message-sign'.")

(defun azos/mail/secure-sign-maybe ()
  "Use `mml-secure-message-sign' unless the addressee is in the
`azos/mail/blacklist-addresses'."
  (let ((addresses (mapcar (lambda (address)
                             (when (string-match
                                    "\\(<?\\)\\([^@< ]+@[^@> ]+\\)\\(>?\\)"
                                    address)
                               (match-string 2 address)))
                           ;; (split-string (message-field-value "To")
                           ;;               "," t "[[:blank:]]+"))))
                           (split-string (concat (message-field-value "To")
                                                 ", "
                                                 (message-field-value "Cc"))
                                         "," t "[[:blank:]]+"))))
    ;; skip secure signing when an address is blacklisted
    (unless (delq nil (mapcar (lambda (address)
                    (car (member address azos/mail/blacklist-addresses)))
                    addresses))
      (mml-secure-message-sign))))
#+end_src

#+begin_src emacs-lisp
(use-package elfeed
    :defer t
    :init
        (defun azos/elfeed/v-mpv (url)
            "Watch a video from URL in MPV"
            (start-process "mpv" nil "mpv" url))

        (defun azos/elfeed/view-mpv (&optional use-generic-p)
        "Youtube-feed link"
        (interactive "P")
        (let ((entries (elfeed-search-selected)))
            (cl-loop for entry in entries
            do (elfeed-untag entry 'unread)
            when (elfeed-entry-link entry)
            do (azos/elfeed/v-mpv it))
        (mapc #'elfeed-search-update-entry entries)))

        (defun slurp (f)
        (with-temp-buffer
            (insert-file-contents f)
            (buffer-substring-no-properties
            (point-min)
            (point-max))))

        (setq-default elfeed-search-filter "@3-days-ago")

        (defun azos/elfeed/open ()
          "Open elfeed in its dedicated \"elfeed\" tab."
          (interactive)
          (azos/tab-bar/open-singleton-tab "elfeed" #'elfeed))
    :config
        (evil-collection-elfeed-setup)

        ;; `global-visual-line-mode' (azos-core/features/base/emacs/config.org)
        ;; overrides elfeed-search-mode's own `truncate-lines' setting, so
        ;; every entry line soft-wraps in anything less than a full-width
        ;; window. Elfeed already wants truncated, not wrapped, lines here.
        ;; `global-visual-line-mode' turns itself on via
        ;; `after-change-major-mode-hook', which runs *after*
        ;; `elfeed-search-mode-hook' - so disabling it there (as a first
        ;; attempt did) toggled visual-line-mode off before it was ever
        ;; turned on. Its off-toggle unconditionally kills the local
        ;; `truncate-lines' value and restores it from a saved snapshot -
        ;; with no prior "on" phase, there's no snapshot, so the kill wins
        ;; and `truncate-lines' is left nil (still soft-wrapping, just at
        ;; character instead of word boundaries). Disabling it from the
        ;; same later hook (past the global mode's own entry) lets the
        ;; snapshot/restore dance complete correctly - but assert
        ;; `truncate-lines' explicitly afterwards anyway, so the outcome
        ;; doesn't depend on that internal save/restore mechanism at all.
        (add-hook 'after-change-major-mode-hook
                  (lambda ()
                    (when (derived-mode-p 'elfeed-search-mode)
                      (visual-line-mode -1)
                      (setq-local truncate-lines t)))
                  100)

        ;; Elfeed renders HTML content (via shr) hard-wrapped to the
        ;; window's width at render time; resizing/splitting afterwards
        ;; leaves the old wrap width in place. Re-render on resize,
        ;; preserving (approximately) where point was so it doesn't jump
        ;; back to the top of the article.
        (defvar-local azos/elfeed-show--last-width nil)
        (defun azos/elfeed-show-refresh-on-resize (&rest _)
          (when (derived-mode-p 'elfeed-show-mode)
            (let ((width (window-body-width)))
              (unless (equal width azos/elfeed-show--last-width)
                (setq azos/elfeed-show--last-width width)
                (let ((pos (point)))
                  (elfeed-show-refresh)
                  (goto-char (min pos (point-max))))))))
        (add-hook 'elfeed-show-mode-hook
                  (lambda ()
                    (add-hook 'window-state-change-hook
                              #'azos/elfeed-show-refresh-on-resize nil t)))
    :bind
        (:map azos/global-minor-mode/open-keymap
              ("e" . azos/elfeed/open))
)
