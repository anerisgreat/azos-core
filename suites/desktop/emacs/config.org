** Mail

*** General instructions

Prerequisites: password-manager entries for each account's passwordCommand set up, mbsync run at least once. Gmail accounts
require an App Password (myaccount.google.com/apppasswords) — not your regular password.

=mu= is auto-initialized by home-manager activation on first deploy. To re-initialize manually:

#+begin_src bash :results none :exports code
mu init --maildir=~/Maildir \
  --my-address=me@example.com \
  --my-address=me.too@example.org
mu index
#+end_src

*** Accounts

Contexts, bookmarks and maildir shortcuts are all built from
=azos/mail/accounts=, which =suites/desktop/mail= generates from
=accounts.email.accounts= into =~/.emacs.d/extra/azos-mail-accounts.el=: one
plist per account, primary first. =:maildir= is the account's maildir
relative to the Maildir root (e.g. ="/work"=); the folder keys are relative
to it.

#+begin_src emacs-lisp
(defvar azos/mail/accounts nil
  "Mail accounts as plists: :name :address :real-name :maildir :inbox :sent
:drafts :trash :refile and, optionally, :spam :exclude :key.")

(defvar azos/mail/maildir-base "~/Maildir"
  "Root of all the accounts' maildirs.")

(defun azos/mail/--folder (account key)
  "ACCOUNT's folder under KEY (e.g. :inbox), as a mu maildir path."
  (concat (plist-get account :maildir) "/" (plist-get account key)))

(defun azos/mail/--maildir-term (folder)
  "A mu query term matching FOLDER, quoted since names can contain spaces."
  (format "maildir:%S" folder))

(defun azos/mail/inbox-query ()
  "mu query for the inboxes of all accounts."
  (concat "("
          (mapconcat (lambda (account)
                       (azos/mail/--maildir-term (azos/mail/--folder account :inbox)))
                     azos/mail/accounts " OR ")
          ") AND NOT flag:trashed"))

(defun azos/mail/--exclusions ()
  "Query suffix leaving out every account's trash, spam and excluded folders."
  (mapconcat
   (lambda (folder) (concat " AND NOT " (azos/mail/--maildir-term folder)))
   (mapcan (lambda (account)
             (mapcar (lambda (f) (concat (plist-get account :maildir) "/" f))
                     (delq nil (append (list (plist-get account :trash)
                                             (plist-get account :spam))
                                       (plist-get account :exclude)))))
           azos/mail/accounts)
   ""))

(defun azos/mail/--in-maildir-p (prefix msg)
  "Whether MSG lives under the maildir PREFIX."
  (and msg (string-prefix-p prefix (mu4e-message-field msg :maildir))))

;; This file has no lexical-binding, so :match-func can't be a closure.
(defun azos/mail/--context (account)
  "A mu4e context for ACCOUNT."
  (make-mu4e-context
   :name (plist-get account :name)
   :match-func (apply-partially #'azos/mail/--in-maildir-p
                                (concat (plist-get account :maildir) "/"))
   :vars `((user-mail-address  . ,(plist-get account :address))
           (user-full-name     . ,(plist-get account :real-name))
           (mu4e-sent-folder   . ,(azos/mail/--folder account :sent))
           (mu4e-drafts-folder . ,(azos/mail/--folder account :drafts))
           (mu4e-trash-folder  . ,(azos/mail/--folder account :trash))
           (mu4e-refile-folder . ,(azos/mail/--folder account :refile)))))
#+end_src

(defvar azos/mail/blacklist-addresses nil
  "List of email addresses (as strings) to blacklist for
  `mml-secure-message-sign'.")

(defun azos/mail/secure-sign-maybe ()
  "Use `mml-secure-message-sign' unless the addressee is in the
`azos/mail/blacklist-addresses'."
  (let ((addresses (mapcar (lambda (address)
                             (when (string-match
                                    "\\(<?\\)\\([^@< ]+@[^@> ]+\\)\\(>?\\)"
                                    address)
                               (match-string 2 address)))
                           ;; (split-string (message-field-value "To")
                           ;;               "," t "[[:blank:]]+"))))
                           (split-string (concat (message-field-value "To")
                                                 ", "
                                                 (message-field-value "Cc"))
                                         "," t "[[:blank:]]+"))))
    ;; skip secure signing when an address is blacklisted
    (unless (delq nil (mapcar (lambda (address)
                    (car (member address azos/mail/blacklist-addresses)))
                    addresses))
      (mml-secure-message-sign))))
#+end_src

*** Send without deadlocking on msmtp

=sendmail-send-it= runs msmtp with the blocking =call-process-region=. If
msmtp's passwordeval calls back into this Emacs (a password helper asking
the running Emacs via emacsclient, or gpg's pinentry served by
pinentry.el), Emacs waits for msmtp while msmtp waits for Emacs, and
=C-c C-c= hangs forever. Run the send process asynchronously and wait while
still serving events. Harmless otherwise.

#+begin_src emacs-lisp
(require 'cl-lib)

(defun azos/mail--call-process-region (start end program &optional delete buffer _display &rest args)
  "Stand-in for `call-process-region' that keeps Emacs processing events."
  (let* ((text (buffer-substring-no-properties start end))
         (out (cond ((eq buffer t) (current-buffer))
                    ((consp buffer) (car buffer))
                    (t buffer)))
         (out (and out (not (eq out 0)) (get-buffer-create out)))
         (proc (make-process :name "azos-sendmail"
                             :buffer out
                             :command (cons program args)
                             :connection-type 'pipe
                             :noquery t
                             :sentinel #'ignore)))
    (when delete
      (delete-region start end))
    (process-send-string proc text)
    (process-send-eof proc)
    (while (process-live-p proc)
      (accept-process-output nil 0.05))
    (process-exit-status proc)))

(defun azos/mail--async-send (orig &rest args)
  "Call ORIG with ARGS, with `call-process-region' made non-blocking."
  (cl-letf (((symbol-function 'call-process-region)
             #'azos/mail--call-process-region))
    (apply orig args)))
#+end_src

*** Unread count in the tab bar

=✉ N= on the tab bar's right group, clickable to open the Inbox bookmark in
mu4e. Opt-in: =azos.mail.tabBarUnreadCount= makes the generated
=azos-mail-accounts.el= call =azos/mail/enable-tab-display=.

It counts the Maildir files of every account's inbox directly rather than
asking mu: the mbsync timer doesn't re-index mu, so its database lags until
mu4e next updates. A message is unread if it is still in =new/=, or in
=cur/= without the =S= (seen) flag; =T= (trashed) ones are skipped.

#+begin_src emacs-lisp
(defvar azos/mail/status-string nil "Holds the unread-mail tab-bar string.")

(defun azos/mail/--unread-in (dir)
  "Number of unread, untrashed messages in the maildir DIR."
  (+ (length (directory-files (expand-file-name "new" dir) nil "\\`[^.]" t))
     (cl-count-if
      (lambda (f)
        (let ((flags (and (string-match ":2,\\([A-Za-z]*\\)\\'" f)
                          (match-string 1 f))))
          (not (and flags (or (string-search "S" flags)
                              (string-search "T" flags))))))
      (directory-files (expand-file-name "cur" dir) nil "\\`[^.]" t))))

(defun azos/mail/--unread-count ()
  "Unread messages across the inboxes of `azos/mail/accounts'."
  (apply #'+ (mapcar (lambda (account)
                       (azos/mail/--unread-in
                        (concat (expand-file-name azos/mail/maildir-base)
                                (azos/mail/--folder account :inbox))))
                     azos/mail/accounts)))

(defun azos/mail/open-inbox ()
  "Show the inboxes of all accounts in mu4e."
  (interactive)
  (require 'mu4e)
  (mu4e-search (azos/mail/inbox-query)))

(defun azos/mail/set-status-string ()
  "Recount unread mail for the tab bar."
  (let ((n (condition-case nil (azos/mail/--unread-count) (error nil))))
    (setq azos/mail/status-string
          (and n
               (concat
                "✉ "
                (propertize (number-to-string n)
                            'face (list :foreground (if (> n 0) "red3" "green3")
                                        :background "LightYellow3"
                                        :box (list :color "DarkSlateGrey"))
                            'local-map (let ((map (make-sparse-keymap)))
                                         (define-key map [tab-bar mouse-1]
                                                     #'azos/mail/open-inbox)
                                         map)
                            'help-echo "mouse-1: open inbox"))))))

(defun azos/mail/get-status-string ()
  "Get the unread-mail tab-bar string."
  (or azos/mail/status-string ""))

(defun azos/mail/enable-tab-display ()
  "Enables unread-mail display in tab bar."
  (cl-pushnew 'azos/mail/get-status-string azos/tab-bar/right-group)
  (cl-pushnew 'azos/mail/set-status-string azos/tab-bar/right-update-group))
#+end_src

#+begin_src emacs-lisp
(use-package elfeed
    :defer t
    :init
        (defun azos/elfeed/v-mpv (url)
            "Watch a video from URL in MPV"
            (start-process "mpv" nil "mpv" url))

        (defun azos/elfeed/view-mpv (&optional use-generic-p)
        "Youtube-feed link"
        (interactive "P")
        (let ((entries (elfeed-search-selected)))
            (cl-loop for entry in entries
            do (elfeed-untag entry 'unread)
            when (elfeed-entry-link entry)
            do (azos/elfeed/v-mpv it))
        (mapc #'elfeed-search-update-entry entries)))

        (defun slurp (f)
        (with-temp-buffer
            (insert-file-contents f)
            (buffer-substring-no-properties
            (point-min)
            (point-max))))

        (setq-default elfeed-search-filter "@3-days-ago")

        (defun azos/elfeed/open ()
          "Open elfeed in its dedicated \"elfeed\" tab."
          (interactive)
          (azos/tab-bar/open-singleton-tab "elfeed" #'elfeed))
    :config
        (evil-collection-elfeed-setup)

        ;; `global-visual-line-mode' (azos-core/suites/base/emacs/config.org)
        ;; overrides elfeed-search-mode's own `truncate-lines' setting, so
        ;; every entry line soft-wraps in anything less than a full-width
        ;; window. Elfeed already wants truncated, not wrapped, lines here.
        ;; `global-visual-line-mode' turns itself on via
        ;; `after-change-major-mode-hook', which runs *after*
        ;; `elfeed-search-mode-hook' - so disabling it there (as a first
        ;; attempt did) toggled visual-line-mode off before it was ever
        ;; turned on. Its off-toggle unconditionally kills the local
        ;; `truncate-lines' value and restores it from a saved snapshot -
        ;; with no prior "on" phase, there's no snapshot, so the kill wins
        ;; and `truncate-lines' is left nil (still soft-wrapping, just at
        ;; character instead of word boundaries). Disabling it from the
        ;; same later hook (past the global mode's own entry) lets the
        ;; snapshot/restore dance complete correctly - but assert
        ;; `truncate-lines' explicitly afterwards anyway, so the outcome
        ;; doesn't depend on that internal save/restore mechanism at all.
        (add-hook 'after-change-major-mode-hook
                  (lambda ()
                    (when (derived-mode-p 'elfeed-search-mode)
                      (visual-line-mode -1)
                      (setq-local truncate-lines t)))
                  100)

        ;; Elfeed renders HTML content (via shr) hard-wrapped to the
        ;; window's width at render time; resizing/splitting afterwards
        ;; leaves the old wrap width in place. Re-render on resize,
        ;; preserving (approximately) where point was so it doesn't jump
        ;; back to the top of the article.
        (defvar-local azos/elfeed-show--last-width nil)
        (defun azos/elfeed-show-refresh-on-resize (&rest _)
          (when (derived-mode-p 'elfeed-show-mode)
            (let ((width (window-body-width)))
              (unless (equal width azos/elfeed-show--last-width)
                (setq azos/elfeed-show--last-width width)
                (let ((pos (point)))
                  (elfeed-show-refresh)
                  (goto-char (min pos (point-max))))))))
        (add-hook 'elfeed-show-mode-hook
                  (lambda ()
                    (add-hook 'window-state-change-hook
                              #'azos/elfeed-show-refresh-on-resize nil t)))
    :bind
        (:map azos/global-minor-mode/open-keymap
              ("e" . azos/elfeed/open))
)
#+end_src
