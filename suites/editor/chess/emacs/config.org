* Lichess

#+begin_src emacs-lisp
(use-package lichess
  :demand t
  :init
  (defun azos/chess/open ()
    "Open lichess in its dedicated \"chess\" tab."
    (interactive)
    (azos/tab-bar/open-singleton-tab "chess" #'lichess))
  :bind
  (:map azos/global-minor-mode/open-keymap
        ("k" . azos/chess/open)))
#+end_src
