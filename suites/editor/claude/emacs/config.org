#+title: Emacs: Claude Code
#+property: header-args :results silent

claude-code-ide runs the real Claude Code CLI in a vterm buffer, connected
to Emacs over MCP (diagnostics, xref, ediff diffs, selection), so the CLI's
own menus, slash commands and =/voice= work as they do in a terminal. It
takes over the =M-o a= agent keys and the tab-bar squares from agent-shell;
agent-shell itself stays available through =M-x agent-shell=.

* Session state

Every session claude-code-ide starts gets =--settings= with hooks that copy
each hook's JSON to =<state dir>/<port>.hook=, where the port is the
session's =CLAUDE_CODE_SSE_PORT=. Hooks given with =--settings= run in
addition to the ones in =~/.claude/settings.json=, so the agent-memory hooks
still fire. Writes go through ="$f.$$"= and =mv=, so Emacs never reads a
half-written file. Session buffers show the thinking spinner's
=✢✳✶✻✽= as =+*×☼¤=: Liberation Mono has no glyph for the stars, and any
fallback font that does is taller, so the spinner line changed height on
every frame. The few other glyphs Claude Code draws that DejaVu Sans Mono
(the base config's monospace fallback for symbols) lacks, =⎿⏺⎯⏸⧉=, are
shown as lookalikes it has, for the same reason: a glyph from a wider font
makes its line wider and the edges of boxes and diffs stop lining up.
After a resize the session repaints (see Rendering below). The session also starts without
=with-editor-export-editor= on =vterm-mode-hook=: claude-code-ide runs the
CLI as the vterm's shell, so the hook's =export EDITOR=...= and =clear=
lines would be typed into Claude's prompt and sent as messages. A file watch
maps the hook event to a state:

- =busy=: a prompt was submitted, or a tool is running
- =blocked=: a permission prompt is showing, or the turn failed with an API error
- =ready=: the turn finished

No hook fires when a turn is interrupted with Escape, so the session stays
=busy= until the next prompt.

The same settings give the CLI a one-line status line: model, effort and
context use (needs =jq=, which the Nix module installs).

(Testing this in =emacs --batch=: file-notify events are only delivered
while waiting in =read-event=; plain =accept-process-output= won't.)

#+begin_src emacs-lisp
(defvar azos/claude/state-dir
  (expand-file-name "azos-claude-ide"
                    (or (getenv "XDG_RUNTIME_DIR") temporary-file-directory))
  "Where Claude Code hooks leave per-session state for Emacs.")

(defvar azos/claude/--buffers (make-hash-table :test #'equal)
  "Session MCP port (a string) -> its Claude Code buffer.")

(defvar azos/claude/--watch nil)

(defvar-local azos/claude/--status 'ready
  "This session's state: `busy', `blocked' or `ready'.")

(defconst azos/claude/--hook-events
  '(SessionStart UserPromptSubmit PreToolUse PostToolUse
    PermissionRequest Notification Stop StopFailure))

(defconst azos/claude/--status-line
  (concat "jq -r '[.model.display_name, .effort.level,"
          " (.context_window.used_percentage"
          " | if . then \"ctx \\(floor)%\" else empty end)]"
          " | map(select(.)) | join(\" · \")'")
  "statusLine command: model · effort · ctx N%.")

(defun azos/claude/--settings-file ()
  "Write the per-session Claude Code settings and return their path."
  (let* ((file (expand-file-name "settings.json" azos/claude/state-dir))
         (write (format "f=%s/$CLAUDE_CODE_SSE_PORT.hook; cat > \"$f.$$\" && mv \"$f.$$\" \"$f\""
                        azos/claude/state-dir))
         (hook `[((hooks . [((type . "command") (command . ,write))]))])
         ;; The · in the status line gets mangled otherwise.
         (coding-system-for-write 'utf-8))
    (make-directory azos/claude/state-dir t)
    (with-temp-file file
      (insert (json-encode
               `((statusLine . ((type . "command")
                                (command . ,azos/claude/--status-line)))
                 (hooks . ,(mapcar (lambda (event) (cons event hook))
                                   azos/claude/--hook-events))))))
    file))

(defun azos/claude/--status-of (data current)
  "The state after hook DATA, given the CURRENT state."
  (pcase (alist-get 'hook_event_name data)
    ((or "UserPromptSubmit" "PreToolUse" "PostToolUse") 'busy)
    ((or "PermissionRequest" "StopFailure") 'blocked)
    ("Notification" (if (equal (alist-get 'notification_type data) "permission_prompt")
                        'blocked
                      current))
    ((or "Stop" "SessionStart") 'ready)
    (_ current)))

(defun azos/claude/--on-change (event)
  (pcase-let* ((`(,_ ,action ,file . ,rest) event)
               (file (if (eq action 'renamed) (car rest) file)))
    (when (and (memq action '(created changed renamed))
               (string-match "/\\([0-9]+\\)\\.hook\\'" file))
      (let ((buffer (gethash (match-string 1 file) azos/claude/--buffers))
            (data (ignore-errors
                    (with-temp-buffer
                      (insert-file-contents file)
                      (json-parse-buffer :object-type 'alist :null-object nil)))))
        (when (and data (buffer-live-p buffer))
          (with-current-buffer buffer
            (setq azos/claude/--status
                  (azos/claude/--status-of data azos/claude/--status)))
          (force-mode-line-update t))))))

(defun azos/claude/--forget (port)
  (remhash port azos/claude/--buffers)
  (ignore-errors (delete-file (expand-file-name (concat port ".hook")
                                                azos/claude/state-dir))))

(defvar azos/claude/--spinner-glyphs
  '((?✢ . ?+) (?✳ . ?*) (?✶ . ?×) (?✻ . ?☼) (?✽ . ?¤))
  "The spinner's stars, and the glyphs the default font has to show instead.")

(defvar azos/claude/--substitute-glyphs
  '((?⎿ . ?└) (?⏺ . ?●) (?⎯ . ?─) (?⏸ . ?‖) (?⧉ . ?❐))
  "Glyphs no monospace font here has, and the lookalikes to show instead.")

(defun azos/claude/--fix-glyphs ()
  "Show the spinner and substitutes as above, keeping vterm's table."
  (let ((table (or buffer-display-table (make-display-table))))
    (pcase-dolist (`(,from . ,to) (append azos/claude/--spinner-glyphs
                                          azos/claude/--substitute-glyphs))
      (aset table from (vector to)))
    (setq buffer-display-table table)))

(defun azos/claude/--create-session (orig buffer-name working-dir port &rest args)
  "Start the session with the state hooks and track its buffer."
  (let* ((vterm-mode-hook (remq 'with-editor-export-editor
                                (bound-and-true-p vterm-mode-hook)))
         (claude-code-ide-cli-extra-flags
          (string-trim (concat claude-code-ide-cli-extra-flags " --settings "
                               (shell-quote-argument (azos/claude/--settings-file)))))
         (result (apply orig buffer-name working-dir port args))
         (key (number-to-string port)))
    (unless (and azos/claude/--watch (file-notify-valid-p azos/claude/--watch))
      (setq azos/claude/--watch
            (file-notify-add-watch azos/claude/state-dir '(change)
                                   #'azos/claude/--on-change)))
    (puthash key (car result) azos/claude/--buffers)
    (with-current-buffer (car result)
      (azos/claude/--fix-glyphs)
      (add-hook 'window-size-change-functions #'azos/claude/--on-resize nil t)
      (add-hook 'kill-buffer-hook (lambda () (azos/claude/--forget key)) nil t))
    result))

(with-eval-after-load 'claude-code-ide
  (advice-add 'claude-code-ide--create-terminal-session :around
              #'azos/claude/--create-session))

(defun azos/claude/buffers ()
  "Live Claude Code session buffers."
  (seq-filter #'buffer-live-p (hash-table-values azos/claude/--buffers)))
#+end_src

* Rendering

After a resize, vterm re-wraps the old screen and Claude Code repaints
over it, which can leave stale fragments behind. Ctrl+L makes Claude Code
clear and repaint the whole screen (the prompt being typed survives), so
it's sent 0.3 s after a session's window stops changing size, and
=M-o a r= sends it by hand. Not in =vterm-copy-mode=, where the key would
leave copy mode instead.

claude-code-ide also keeps height-only resizes from reaching the terminal
(=claude-code-ide-prevent-reflow-glitch=, a workaround for an old Claude
Code scrolling bug). Claude Code now draws only the visible screen, so the
workaround just leaves it drawing at the wrong height. With the variable
nil, claude-code-ide never installs its reflow-filter advice.

#+begin_src emacs-lisp
(setq claude-code-ide-prevent-reflow-glitch nil)

(defvar-local azos/claude/--redraw-timer nil)

(defun azos/claude/redraw (&optional buffer)
  "Make the Claude Code session in BUFFER repaint.
BUFFER defaults to the current session buffer, else the first session."
  (interactive)
  (let ((buffer (or buffer
                    (car (memq (current-buffer) (azos/claude/buffers)))
                    (car (azos/claude/buffers)))))
    (when (and (buffer-live-p buffer) (get-buffer-process buffer))
      (with-current-buffer buffer
        (unless (bound-and-true-p vterm-copy-mode)
          (vterm-send-key "l" nil nil t))))))

(defun azos/claude/--on-resize (window)
  "Redraw WINDOW's session shortly after its size stops changing."
  (with-current-buffer (window-buffer window)
    (when (timerp azos/claude/--redraw-timer)
      (cancel-timer azos/claude/--redraw-timer))
    (setq azos/claude/--redraw-timer
          (run-with-timer 0.3 nil #'azos/claude/redraw (current-buffer)))))

(define-key azos/llm-keymap (kbd "r") #'azos/claude/redraw)
#+end_src

* Tabs and switching

Like agent-shell sessions, a Claude Code session lives in its project's tab
(=azos/agent-shell/tab-name=: the project's name, or =agents=). Inside the
tab, claude-code-ide keeps managing its own side window, which its ediff
and toggle handling expect.

#+begin_src emacs-lisp
(defun azos/claude/display (buffer)
  "Go to BUFFER's project tab and select its Claude Code window."
  (require 'claude-code-ide)
  (require 'agent-shell)
  (azos/tab-bar/open-singleton-tab
   (azos/agent-shell/tab-name buffer)
   (lambda ()
     (select-window (or (get-buffer-window buffer 0)
                        (claude-code-ide--display-buffer-in-side-window buffer))))))

(defun azos/claude/start (&optional new)
  "Jump to this project's most recent Claude Code session, or start one.
With a prefix argument NEW, always start a new session."
  (interactive "P")
  (require 'claude-code-ide)
  (require 'agent-shell)
  (let* ((dir (claude-code-ide--get-working-directory))
         (session (and (not new) (claude-code-ide-mcp--mru-session dir)))
         (buffer (and session (claude-code-ide-mcp-session-buffer session))))
    (if (buffer-live-p buffer)
        (azos/claude/display buffer)
      (azos/tab-bar/open-singleton-tab
       (azos/agent-shell/tab-name (current-buffer))
       (lambda ()
         (let ((default-directory dir))
           (claude-code-ide)))))))

(defun azos/claude/switch-session ()
  "Pick a Claude Code session and jump to it."
  (interactive)
  (let ((candidates
         (mapcar (lambda (buffer)
                   (cons (propertize (buffer-name buffer)
                                     'face (azos/agent-shell/status-face
                                            (buffer-local-value 'azos/claude/--status buffer)))
                         buffer))
                 (azos/claude/buffers))))
    (unless candidates
      (user-error "No Claude Code sessions"))
    (azos/claude/display
     (cdr (assoc (completing-read "Claude Code session: " candidates nil t)
                 candidates)))))

(define-key azos/llm-keymap (kbd "s") #'azos/claude/start)
(define-key azos/llm-keymap (kbd "S") #'azos/claude/switch-session)
(define-key azos/llm-keymap (kbd "i") #'claude-code-ide-menu)
#+end_src

* Tab bar

One square per Claude Code session, in agent-shell's status colors, each
clickable. They replace the agent-shell squares.

#+begin_src emacs-lisp
(defun azos/tab-bar/claude-status-string ()
  "A row of colored squares, one per live Claude Code session."
  (mapconcat
   (lambda (buffer)
     (let ((status (buffer-local-value 'azos/claude/--status buffer)))
       (propertize "■"
                   'face (azos/agent-shell/status-face status)
                   'local-map (let ((map (make-sparse-keymap)))
                                (define-key map [tab-bar mouse-1]
                                  (lambda ()
                                    (interactive)
                                    (azos/claude/display buffer)))
                                map)
                   'help-echo (format "%s (%s)" (buffer-name buffer) status))))
   (azos/claude/buffers)
   ""))

(setq azos/tab-bar/right-group
      (delq 'azos/tab-bar/agent-shell-status-string azos/tab-bar/right-group))
(cl-pushnew 'azos/tab-bar/claude-status-string azos/tab-bar/right-group)

* Spoken replies

With =azos.claude.speak= on, the azos-speak server reads replies aloud.
=azos/claude/stop-speech= silences it, sending the same line as
=azos-speak stop=. Typing Space into a session also does: in voice mode
=tap=, Space starts dictation, and talking back should cut off the spoken
reply. The Nix module sets =azos/claude/speak-enabled= in
=~/.emacs.d/extra/azos-claude.el=, which loads after this package, so the
hook checks it when it runs.

#+begin_src emacs-lisp
(defvar azos/claude/speak-enabled nil
  "Non-nil when azos.claude.speak is on; set by the Nix module.")

(defun azos/claude/stop-speech ()
  "Silence whatever azos-speak is playing; do nothing if it isn't running."
  (condition-case nil
      (let ((proc (make-network-process
                   :name "azos-speak-stop" :family 'local :noquery t
                   :service (expand-file-name
                             "azos-speak/server.sock"
                             (or (getenv "XDG_RUNTIME_DIR")
                                 (format "/run/user/%d" (user-uid)))))))
        (process-send-string proc "{\"cmd\": \"stop\"}\n")
        (delete-process proc))
    (error nil)))

(defun azos/claude/--stop-speech-on-space ()
  "Stop speech when Space is typed into a Claude Code session."
  (when (and azos/claude/speak-enabled
             (eq this-command 'vterm--self-insert)
             (equal (this-command-keys) " ")
             (memq (current-buffer) (azos/claude/buffers)))
    (azos/claude/stop-speech)))

(add-hook 'pre-command-hook #'azos/claude/--stop-speech-on-space)
#+end_src
